# --- Self-elevate to Administrator ---
if (-NOT ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)) {
    Write-Host "Elevating to Administrator..." -ForegroundColor Yellow
    $scriptPath = $MyInvocation.MyCommand.Definition
    if ($scriptPath -like "*toolkit.ps1*" -or $scriptPath -like "*toolkit.txt*") {
        Start-Process powershell -Verb RunAs -ArgumentList "-NoProfile -ExecutionPolicy Bypass -File `"$scriptPath`""
    } else {
        Start-Process powershell -Verb RunAs -ArgumentList "-NoProfile -ExecutionPolicy Bypass -Command `"irm 'https://raw.githubusercontent.com/bizarriahuz/it-toolkit/main/toolkit.txt' | iex`""
    }
    return
}

# --- Helper functions ---
function Write-Header($text) {
    Clear-Host
    Write-Host "===============================================" -ForegroundColor Cyan
    Write-Host "  $text" -ForegroundColor Cyan
    Write-Host "===============================================" -ForegroundColor Cyan
    Write-Host ""
}
function Write-OK($msg)   { Write-Host "[ OK ] $msg" -ForegroundColor Green }
function Write-Warn($msg) { Write-Host "[WARN] $msg" -ForegroundColor Yellow }
function Write-Err($msg)  { Write-Host "[FAIL] $msg" -ForegroundColor Red }

# --- Setup & Deployment ---
function Install-Action1 {
    Write-Header "Install Action1 Agent"
    $msiUrl = "https://app.na-2.action1.com/agent/21c1cc54-c786-11f1-9f08-c1a2c9c50f19/windows/agent(New_Endpoints).msi" 
    $msiPath = "$env:TEMP\action1_agent.msi"
    try {
        Write-Host "Downloading MSI (Requires Internet)..."
        Invoke-WebRequest -Uri $msiUrl -OutFile $msiPath -UseBasicParsing
        Write-Host "Installing silently..."
        Start-Process msiexec.exe -ArgumentList "/i `"$msiPath`" /quiet /qn /norestart" -Wait
        Write-OK "Action1 agent installed."
    } catch { Write-Err $_.Exception.Message }
    Pause
}

function Remove-Atera {
    Write-Header "Remove Atera Agent"
    $paths = @(
        "$env:ProgramFiles\ATERA Networks",
        "${env:ProgramFiles(x86)}\ATERA Networks"
    )
    $found = $paths | Where-Object { Test-Path $_ }
    if (-not $found) { Write-Warn "Atera not found on this machine."; Pause; return }

    Get-Service -Name "Atera*" -ErrorAction SilentlyContinue | ForEach-Object {
        Write-Host "Stopping service: $($_.Name)"
        Stop-Service $_.Name -Force -ErrorAction SilentlyContinue
    }
    Get-Process -Name "Atera*" -ErrorAction SilentlyContinue | Stop-Process -Force -ErrorAction SilentlyContinue
    $found | ForEach-Object {
        Write-Host "Removing: $_"
        Remove-Item $_ -Recurse -Force -ErrorAction SilentlyContinue
    }
    Write-OK "Atera cleanup complete. Reboot recommended."
    Pause
}

function Install-StandardApps {
    Write-Header "Install Standard Apps (via Winget)"
    if (-not (Get-Command winget -ErrorAction SilentlyContinue)) {
        Write-Err "Winget is not installed on this machine."
        Pause
        return
    }
    
    $apps = @(
        "Google.Chrome",
        "7zip.7zip",
        "Adobe.Acrobat.Reader.64-bit",
        "Notepad++.Notepad++"
    )
    
    foreach ($app in $apps) {
        Write-Host "Installing $app..." -ForegroundColor Yellow
        winget install --id $app --silent --accept-package-agreements --accept-source-agreements
    }
    Write-OK "App installation complete."
    Pause
}

# --- System & Troubleshooting ---
function Get-SystemHealth {
    Write-Header "System Health Report"
    $os     = Get-CimInstance Win32_OperatingSystem
    $cs     = Get-CimInstance Win32_ComputerSystem
    $disk   = Get-CimInstance Win32_LogicalDisk -Filter "DeviceID='C:'"
    $batt   = Get-CimInstance Win32_Battery -ErrorAction SilentlyContinue
    $defender = Get-MpComputerStatus -ErrorAction SilentlyContinue

    Write-Host "Hostname      : $($cs.Name)"
    Write-Host "Model         : $($cs.Manufacturer) $($cs.Model)"
    Write-Host "OS            : $($os.Caption) $($os.Version)"
    Write-Host "Last Boot     : $($os.LastBootUpTime)"
    Write-Host ("C: Free       : {0:N2} GB / {1:N2} GB" -f ($disk.FreeSpace/1GB), ($disk.Size/1GB))
    Write-Host "Pending Reboot: $([bool](Test-Path 'HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\WindowsUpdate\Auto Update\RebootRequired'))"
    if ($batt) { Write-Host "Battery       : $($batt.EstimatedChargeRemaining)% remaining" }
    if ($defender) {
        Write-Host "Defender AV   : RealTime=$($defender.RealTimeProtectionEnabled) | SignatureAge=$($defender.AntivirusSignatureAge)d"
    }
    Pause
}

function Repair-DomainTrust {
    Write-Header "Repair Domain Trust Relationship"
    $cs = Get-CimInstance Win32_ComputerSystem
    if ($cs.PartOfDomain -eq $false) {
        Write-Err "This computer is not joined to a domain."
        Pause
        return
    }

    Write-Host "Computer Name : $($cs.Name)"
    Write-Host "Domain        : $($cs.Domain)"
    Write-Host ""
    Write-Host "Testing connection to domain controller..." -NoNewline
    if (Test-ComputerSecureChannel -ErrorAction SilentlyContinue) {
        Write-OK " Secure channel is currently HEALTHY. No repair needed."
        Pause
        return
    } else {
        Write-Warn " Secure channel is BROKEN."
    }

    Write-Host ""
    Write-Host "This will attempt to reset the machine password in Active Directory." -ForegroundColor Yellow
    Write-Host "You MUST provide Domain Admin credentials to proceed." -ForegroundColor Yellow
    Write-Host ""
    $confirm = Read-Host "Do you want to attempt the repair? (Y/N)"
    if ($confirm -match '^[Yy]') {
        try {
            Write-Host "Attempting repair..." -NoNewline
            $result = Test-ComputerSecureChannel -Repair -Credential (Get-Credential) -ErrorAction Stop
            if ($result) {
                Write-OK " Success!"
                Write-Host "Please restart the computer for changes to take effect." -ForegroundColor Green
            }
        } catch {
            Write-Err " Failed."
            Write-Host $_.Exception.Message -ForegroundColor Red
            Write-Host "If this failed, the computer may need to be removed and re-joined to the domain manually." -ForegroundColor Yellow
        }
    } else {
        Write-Host "Repair cancelled."
    }
    Pause
}

function Invoke-GPUpdate {
    Write-Header "Force Group Policy Update"
    Write-Host "Running gpupdate /force..."
    Start-Process gpupdate.exe -ArgumentList "/force" -Wait -NoNewWindow
    Write-OK "Group Policy updated."
    Pause
}

function Reset-WindowsUpdate {
    Write-Header "Reset Windows Update Cache"
    Write-Host "Stopping services..." -NoNewline
    Stop-Service -Name wuauserv -Force -ErrorAction SilentlyContinue
    Stop-Service -Name bits -Force -ErrorAction SilentlyContinue
    Write-OK " Done."
    
    Write-Host "Clearing SoftwareDistribution..." -NoNewline
    Remove-Item "C:\Windows\SoftwareDistribution\*" -Recurse -Force -ErrorAction SilentlyContinue
    Write-OK " Done."
    
    Write-Host "Restarting services..." -NoNewline
    Start-Service -Name wuauserv -ErrorAction SilentlyContinue
    Start-Service -Name bits -ErrorAction SilentlyContinue
    Write-OK " Done."
    
    Write-Host "Windows Update cache reset. Try checking for updates again." -ForegroundColor Green
    Pause
}

function Repair-WindowsSystem {
    Write-Header "Scan & Repair Windows System Files"
    Write-Warn "This process can take 15-30 minutes. Do not close the window."
    $confirm = Read-Host "Do you want to proceed? (Y/N)"
    if ($confirm -match '^[Yy]') {
        Write-Host "`n1. Running System File Checker (SFC)..." -ForegroundColor Yellow
        sfc /scannow
        
        Write-Host "`n2. Running DISM Image Repair..." -ForegroundColor Yellow
        DISM /Online /Cleanup-Image /RestoreHealth
        
        Write-OK "Repair process complete. Reboot recommended."
    } else {
        Write-Host "Cancelled."
    }
    Pause
}

function Get-TopProcesses {
    Write-Header "Top 5 Resource-Hogging Processes"
    Write-Host "--- Top 5 by CPU ---" -ForegroundColor Cyan
    Get-Process | Sort-Object CPU -Descending | Select-Object -First 5 Name, @{Name="CPU(s)";Expression={[math]::Round($_.CPU, 2)}} | Format-Table -AutoSize
    
    Write-Host "--- Top 5 by RAM (Working Set) ---" -ForegroundColor Cyan
    Get-Process | Sort-Object WorkingSet -Descending | Select-Object -First 5 Name, @{Name="RAM(MB)";Expression={[math]::Round($_.WorkingSet / 1MB, 2)}} | Format-Table -AutoSize
    Pause
}

# --- Network & Hardware ---
function Get-NetworkAndSessions {
    Write-Header "User Sessions & IP Addresses"
    Write-Host "--- Active User Sessions ---" -ForegroundColor Cyan
    query user 2>$null
    if (-not $?) { Write-Host "No interactive users logged in." }
    
    Write-Host "`n--- IPv4 Addresses ---" -ForegroundColor Cyan
    Get-NetIPAddress -AddressFamily IPv4 | Where-Object { $_.IPAddress -notlike "127.*" -and $_.IPAddress -notlike "169.254.*" } | Select-Object InterfaceAlias, IPAddress | Format-Table -AutoSize
    Pause
}

function Reset-Network {
    Write-Header "Flush DNS & Reset Network"
    Write-Host "Flushing DNS cache..." -NoNewline
    ipconfig /flushdns | Out-Null
    Write-OK " Done."
    Write-Host "Renewing IP address..." -NoNewline
    ipconfig /release | Out-Null
    ipconfig /renew | Out-Null
    Write-OK " Done."
    Write-Host "Registering DNS..." -NoNewline
    ipconfig /registerdns | Out-Null
    Write-OK " Done."
    Write-Host "Network reset complete." -ForegroundColor Green
    Pause
}

function Get-WiFiPasswords {
    Write-Header "Saved Wi-Fi Passwords"
    $profiles = netsh wlan show profiles | Select-String "All User Profile" | ForEach-Object { $_.ToString().Split(":")[1].Trim() }
    
    if (-not $profiles) {
        Write-Warn "No saved Wi-Fi profiles found."
        Pause
        return
    }

    foreach ($profile in $profiles) {
        $details = netsh wlan show profile name="$profile" key=clear
        $passwordLine = $details | Select-String "Key Content" 
        
        if ($passwordLine) {
            $password = $passwordLine.ToString().Split(":")[1].Trim()
            Write-Host "Network: " -NoNewline; Write-Host $profile -ForegroundColor Cyan -NoNewline
            Write-Host " | Password: " -NoNewline; Write-Host $password -ForegroundColor Green
        } else {
            Write-Host "Network: " -NoNewline; Write-Host $profile -ForegroundColor Cyan -NoNewline
            Write-Host " | Password: [Open/No Password]" -ForegroundColor Gray
        }
    }
    Pause
}

function Get-DiskHealth {
    Write-Header "Physical Disk Health (SMART)"
    $disks = Get-PhysicalDisk
    foreach ($disk in $disks) {
        Write-Host "Drive: $($disk.FriendlyName)" -ForegroundColor Cyan
        if ($disk.HealthStatus -eq "Healthy") {
            Write-OK " Status: Healthy"
        } else {
            Write-Err " Status: $($disk.HealthStatus) - WARNING!"
        }
        Write-Host " Media Type: $($disk.MediaType)"
        Write-Host ""
    }
    Pause
}

function Get-DriveWear {
    Write-Header "Drive Wear & Tear (SSD/HDD)"
    $disks = Get-PhysicalDisk
    foreach ($disk in $disks) {
        Write-Host "Drive: $($disk.FriendlyName)" -ForegroundColor Cyan
        $wear = $disk | Get-StorageReliabilityCounter | Select-Object -ExpandProperty Wear
        
        if ($null -ne $wear) {
            if ($wear -gt 80) {
                Write-Err " Wear Level: $wear% - CRITICAL! Drive needs replacing soon."
            } elseif ($wear -gt 50) {
                Write-Warn " Wear Level: $wear% - Keep an eye on this one."
            } else {
                Write-OK " Wear Level: $wear% - Drive is in great shape."
            }
        } else {
            Write-Host " Wear Level: Not supported by this drive." -ForegroundColor Gray
        }
        Write-Host ""
    }
    Pause
}

function Get-BatteryReport {
    Write-Header "Generate Battery Health Report"
    $reportPath = "$env:USERPROFILE\battery-report.html"
    Write-Host "Generating report..." -NoNewline
    powercfg /batteryreport /output $reportPath | Out-Null
    Write-OK " Done."
    Write-Host "Report saved to: $reportPath" -ForegroundColor Green
    Write-Host "Open this file in a browser to see battery design capacity vs. current capacity." -ForegroundColor Yellow
    Pause
}

function Get-EncryptionStatus {
    Write-Header "BitLocker & Activation Status"
    
    Write-Host "--- BitLocker Status ---" -ForegroundColor Cyan
    try {
        $bl = Get-BitLockerVolume -MountPoint "C:" -ErrorAction Stop
        Write-Host "Volume C: Status : $($bl.ProtectionStatus)"
        Write-Host "Encryption %     : $($bl.EncryptionPercentage)%"
        if ($bl.ProtectionStatus -eq "Off") {
            Write-Warn "BitLocker is NOT enabled on C:!"
        } else {
            Write-OK "BitLocker protection is active."
        }
    } catch {
        Write-Warn "BitLocker cmdlets not available (requires Windows Pro/Enterprise)."
    }

    Write-Host "`n--- Windows Activation ---" -ForegroundColor Cyan
    try {
        $status = Get-CimInstance -ClassName SoftwareLicensingProduct -Filter "PartialProductKey IS NOT NULL AND LicenseStatus=1"
        if ($status) {
            Write-OK "Windows is activated."
            Write-Host "Product: $($status.Name)"
        } else {
            Write-Warn "Windows is NOT activated."
        }
    } catch {
        Write-Warn "Could not determine activation status."
    }
    Pause
}

# --- Utility ---
function Invoke-Cleanup {
    Write-Header "System Cleanup"
    $before = (Get-PSDrive C).Free
    Remove-Item "$env:TEMP\*" -Recurse -Force -ErrorAction SilentlyContinue
    Remove-Item "C:\Windows\Temp\*" -Recurse -Force -ErrorAction SilentlyContinue
    Clear-RecycleBin -Force -ErrorAction SilentlyContinue
    $after = (Get-PSDrive C).Free
    $freed = [math]::Round(($after - $before)/1MB, 2)
    Write-OK "Freed approximately $freed MB."
    Pause
}

function Reset-PrintSpooler {
    Write-Header "Clear Print Spooler Queue"
    Write-Host "Stopping Print Spooler service..." -NoNewline
    Stop-Service -Name Spooler -Force -ErrorAction SilentlyContinue
    Write-OK " Stopped."
    Write-Host "Clearing stuck print jobs..." -NoNewline
    Remove-Item "C:\Windows\System32\spool\PRINTERS\*" -Force -ErrorAction SilentlyContinue
    Write-OK " Cleared."
    Write-Host "Starting Print Spooler service..." -NoNewline
    Start-Service -Name Spooler -ErrorAction SilentlyContinue
    Write-OK " Started."
    Write-Host "Printer queue reset. Try printing again." -ForegroundColor Green
    Pause
}

function Remove-Bloatware {
    Write-Header "Remove Standard Windows Bloatware"
    $bloat = @(
        "Microsoft.BingNews", "Microsoft.BingWeather", "Microsoft.GamingApp",
        "Microsoft.GetHelp", "Microsoft.MicrosoftSolitaireCollection",
        "Microsoft.ZuneMusic", "Microsoft.ZuneVideo", "Microsoft.XboxApp"
    )
    foreach ($app in $bloat) {
        Write-Host "Removing $app..." -NoNewline
        Get-AppxPackage -Name $app -AllUsers | Remove-AppxPackage -AllUsers -ErrorAction SilentlyContinue
        Write-OK " Done."
    }
    Write-Host "Bloatware removed." -ForegroundColor Green
    Pause
}

function Get-UserInfo {

function Restart-ToBIOS {
    Write-Header "Restart to BIOS/UEFI"
    Write-Warn "This will restart the computer immediately and boot into the BIOS."
    $confirm = Read-Host "Are you sure? (Y/N)"
    if ($confirm -match '^[Yy]') {
        shutdown.exe /r /fw /t 0
    } else {
        Write-Host "Cancelled."
    }
}
